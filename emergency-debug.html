🚨 ЭКСТРЕННАЯ ОТЛАДКА TELEGRAM WEBAPP

⚠️ КРИТИЧНО: Вставьте код в консоль Telegram WebApp!

📋 ШАГИ:

  1. Откройте https://web.telegram.org
  2. Найдите бота и откройте WebApp
  3. Нажмите F12 → вкладка Console
  4. Вставьте ВЕСЬ этот код:
// ЭКСТРЕННЫЙ ДЕБАГ КОД - ВСТАВИТЬ В КОНСОЛЬ console.log('🚨 === ЭКСТРЕННАЯ ОТЛАДКА ==='); console.log('📍 URL:', window.location.href); console.log('🔍 Telegram object:', !!window.Telegram); console.log('📱 WebApp object:', !!window.Telegram?.WebApp); if (window.Telegram?.WebApp) { console.log('✅ SUCCESS: Telegram WebApp доступен!'); console.log('👤 User:', window.Telegram.WebApp.initDataUnsafe?.user?.first_name); console.log('📤 sendData:', typeof window.Telegram.WebApp.sendData); console.log('🔒 close:', typeof window.Telegram.WebApp.close); // Тестовые команды window.DEBUG = { testSend: () => { console.log('📤 Testing sendData...'); window.Telegram.WebApp.sendData(JSON.stringify({test: 'emergency_debug'})); }, testClose: () => { console.log('🔒 Testing close...'); window.Telegram.WebApp.close(); }, testApp: () => { console.log('🎯 Testing full app flow...'); // Имитация того, что делает кнопка const data = { service_type: "bot_development_order", action: "order_bot_development", quiz_result: { bot_type: "LINEAR", title: "Test Bot" } }; console.log('📦 Sending:', data); window.Telegram.WebApp.sendData(JSON.stringify(data)); setTimeout(() => window.Telegram.WebApp.close(), 1000); } }; console.log('🛠️ Команды: DEBUG.testSend(), DEBUG.testClose(), DEBUG.testApp()'); } else { console.log('❌ FAIL: Telegram WebApp НЕДОСТУПЕН!'); console.log('🔍 Ищем Telegram API...'); // Поиск в window for (let key in window) { if (key.toLowerCase().includes('telegram')) { console.log('📍 Found:', key, '=', window[key]); } } console.log('💡 Убедитесь, что вы в WebApp бота!'); }
